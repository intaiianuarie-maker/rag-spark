async def verification(use_verifier):
    calls = []
    class Recorder:
        async def verify_claim(self, claim_text, evidence, *, recheck=False):
            calls.append(claim_text)
            return ClaimVerdict(claim_id=claim_text, verdict=Verdict.SUPPORTED,
                                evidence_ids=[], reason_code='SUPPORTED')
    evidence = {
        'claims': [Claim(text='The launch budget is 20 dollars.', chunk_ids=['ch-1'])],
        'canonical': {'ch-1': CanonicalChunk(chunk_id='ch-1', workspace_id='ws',
            document_id='doc-1', content='The launch budget is 20 dollars.')},
        'allowed_workspace_ids': {'ws'},
    }
    if use_verifier:
        evidence['verifier'] = Recorder()
    provider = AsyncMock()
    provider.generate.return_value = GenerationResult(
        text='The moon is made of cheese.', model='google/gemma-4-26b-a4b-it:free')
    node = NodeSpec(id='llm', type=NodeType.LLM,
        config={'requires_verification': True, 'evidence': evidence},
        required_capabilities=['llm'])
    result = await ProductionNodeExecutor(llm_provider=provider).execute_node(
        node, 'Check facts.', 'ws')
    return {'result': result, 'semantic_verifier_calls': calls}


async def provenance(path):
    store = SQLiteGraphStore(path)
    docs = [str(uuid.uuid4()), str(uuid.uuid4())]
    chunks = [str(uuid.uuid4()), str(uuid.uuid4())]
    try:
        for doc, chunk in zip(docs, chunks):
            ent = Entity(id='alice', name='Alice Smith', workspace_id='ws',
                         document_ids=[doc], chunk_ids=[chunk])
            await store.upsert_extraction('ws', [ent], [], [], document_id=doc)
        await store.tombstone_document('ws', docs[1])
        ent = await store.get_entity('ws', 'alice')
        search = await GraphRetriever(store).local_search('ws', 'Alice')
        return {'remaining_document_ids': ent.document_ids,
                'remaining_chunk_ids': ent.chunk_ids,
                'deleted_chunk_id': chunks[1],
                'deleted_chunk_in_graph_search': chunks[1] in search.evidence_chunk_ids}
    finally:
        store.close()


async def stale_version(path):
    store = SQLiteGraphStore(path)
    try:
        extractor = RuleBasedGraphExtractor()
        old = await extractor.extract([GraphChunk(chunk_id='old-version-chunk',
            text='Alice Smith works at Acme Corp.', workspace_id='ws', document_id='doc')])
        await store.upsert_extraction('ws', old.entities, old.relationships, [], document_id='doc')
        await store.tombstone_chunks('ws', ['old-version-chunk'])
        absent = await store.get_entity('ws', old.entities[0].id) is None
        counts = await store.upsert_extraction('ws', old.entities, old.relationships, [], document_id='doc')
        search = await GraphRetriever(store).local_search('ws', 'Alice')
        return {'absent_after_supersession': absent, 'stale_upsert_counts': counts,
                'old_chunk_returns_after_stale_extraction': 'old-version-chunk' in search.evidence_chunk_ids}
    finally:
        store.close()


async def deletion_race(path):
    store = SQLiteGraphStore(path)
    conn = store._connect()
    fired = []
    class FrozenCursor:
        def __init__(self, row):
            self.row = row
        def fetchone(self):
            return self.row
    class ConnectionProxy:
        def __enter__(self):
            conn.__enter__()
            return self
        def __exit__(self, *args):
            return conn.__exit__(*args)
        def __getattr__(self, name):
            return getattr(conn, name)
        def execute(self, sql, args=()):
            cur = conn.execute(sql, args)
            if not fired and sql.startswith('SELECT tombstoned FROM graph_tombstones'):
                row = cur.fetchone()
                fired.append(True)
                child = subprocess.run([sys.executable, __file__, '--delete', str(path)],
                    capture_output=True, text=True, timeout=15, check=True)
                return FrozenCursor(row)
            return cur
    store._connect = lambda: ConnectionProxy()
    try:
        ent = Entity(id='fresh-after-delete', name='Alice', workspace_id='ws',
                     document_ids=['doc'], chunk_ids=['new-chunk'])
        counts = await store.upsert_extraction('ws', [ent], [], [], document_id='doc')
        return {'independent_deletion_completed_between_check_and_begin': bool(fired),
                'document_tombstoned': await store.is_document_tombstoned('ws', 'doc'),
                'stale_upsert_counts': counts,
                'live_entity_after_document_delete': await store.get_entity('ws', ent.id) is not None}
    finally:
        store.close()


async def main():
    with tempfile.TemporaryDirectory(prefix='super-rag-review-') as root:
        p = Path(root)
        queue = get_queue()
        await queue.enqueue('review-run')
        child = subprocess.run([sys.executable, __file__, '--queue-depth'],
            capture_output=True, text=True, timeout=15, check=True)
        return {'default_agent_queue': {'parent_depth': await queue.depth(),
                    'independent_worker_depth': int(child.stdout)},
                'workflow_no_semantic_verifier': await verification(False),
                'workflow_output_unchecked_with_verifier': await verification(True),
                'graph_delete_provenance': await provenance(p / 'provenance.sqlite'),
                'graph_stale_version': await stale_version(p / 'version.sqlite'),
                'graph_delete_transaction_race': await deletion_race(p / 'race.sqlite')}
